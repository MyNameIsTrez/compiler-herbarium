func println1(x any) {
    print(x)
}

func Println(a []any) {
    sink = a[0]
    i := 0
    for i < len(a) {
        println1(a[i])
        i = i + 1
    }
}

func main() {
    sum := any(0)
    i := any(1)
    for i.(int) < 11 {
        sum = any(sum.(int) + i.(int))
        i = any(i.(int) + 1)
    }
    Println([]any{sum})
}
