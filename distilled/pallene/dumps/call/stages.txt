== stage_source: the program of the example ==
function fact(n: integer): integer
    if n < 2 then
        return 1
    end
    return n * fact(n - 1)
end
function main()
    local r: integer = fact(5)
    local s: string = tostring(r) .. "\n"
    io.write(s)
end

== stage_to_ir: basic blocks of flat commands ==
function fact
  variables: n:integer $t1:integer $t2:boolean $t3:integer $t4:integer $t5:integer
  B0:
    $t2 = n < 2
    if $t2 goto B1 else B3
  B1:
    $t1 = 1
    goto B4
  B2:
    goto B3
  B3:
    $t5 = n - 1
    $t4 = CallStatic fact($t5)
    $t3 = n * $t4
    $t1 = $t3
    goto B4
  B4:
function main
  variables: r:integer s:string $t2:string $t3:any
  B0:
    r = CallStatic fact(5)
    $t3 = ToDyn<integer>(r)
    $t2 = BuiltinTostring($t3)
    s = $t2 .. "\n"
    CheckGC
    BuiltinIoWrite(s)
