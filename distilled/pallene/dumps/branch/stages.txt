== stage_source: the program of the example ==
function main()
    local a: integer = 3
    local b: integer = 5
    local s: string = ""
    if a < b then
        s = tostring(b)
    else
        s = tostring(a)
    end
    local line: string = s .. "\n"
    io.write(line)
end
