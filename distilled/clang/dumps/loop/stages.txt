== stage_irgen: IR as clang emits it, every local in an alloca ==
define void @main() {
entry:
  %sum = alloca i64
  %i = alloca i64
  store i64 0, ptr %sum
  store i64 1, ptr %i
  br label %while.cond
while.cond:
  %0 = load i64, ptr %i
  %cmp = icmp slt i64 %0, 11
  br i1 %cmp, label %while.body, label %while.end
while.body:
  %1 = load i64, ptr %sum
  %2 = load i64, ptr %i
  %add = add i64 %1, %2
  store i64 %add, ptr %sum
  %3 = load i64, ptr %i
  %add1 = add i64 %3, 1
  store i64 %add1, ptr %i
  br label %while.cond
while.end:
  %4 = load i64, ptr %sum
  call void @print_int(i64 %4)
  ret void
}

define void @main() {
entry:
  br label %while.cond
while.cond:
  %sum.0 = phi i64 [ 0, %entry ], [ %add, %while.body ]
  %i.0 = phi i64 [ 1, %entry ], [ %add1, %while.body ]
  %cmp = icmp slt i64 %i.0, 11
  br i1 %cmp, label %while.body, label %while.end
while.body:
  %add = add i64 %sum.0, %i.0
  %add1 = add i64 %i.0, 1
  br label %while.cond
while.end:
  call void @print_int(i64 %sum.0)
  ret void
}
