== stage_irgen: IR as clang emits it, every local in an alloca ==
define i64 @fact(i64 %n) {
entry:
  %retval = alloca i64
  %n.addr = alloca i64
  store i64 %n, ptr %n.addr
  %0 = load i64, ptr %n.addr
  %cmp = icmp slt i64 %0, 2
  br i1 %cmp, label %if.then, label %if.end
if.then:
  store i64 1, ptr %retval
  br label %return
if.end:
  %1 = load i64, ptr %n.addr
  %2 = load i64, ptr %n.addr
  %sub = sub i64 %2, 1
  %call = call i64 @fact(i64 %sub)
  %mul = mul i64 %1, %call
  store i64 %mul, ptr %retval
  br label %return
return:
  %3 = load i64, ptr %retval
  ret i64 %3
}
define void @main() {
entry:
  %call = call i64 @fact(i64 5)
  call void @print_int(i64 %call)
  ret void
}
== stage_mem2reg: what it did ==
fact: immediate dominators: entry<-entry if.then<-entry if.end<-entry return<-entry
%retval: stores in: if.then if.end; frontier: return; live-in: return; phis in: return
%n.addr: one store, loads replaced by the stored value
main: immediate dominators: entry<-entry

define i64 @fact(i64 %n) {
entry:
  %cmp = icmp slt i64 %n, 2
  br i1 %cmp, label %if.then, label %if.end
if.then:
  br label %return
if.end:
  %sub = sub i64 %n, 1
  %call = call i64 @fact(i64 %sub)
  %mul = mul i64 %n, %call
  br label %return

  %retval.0 = phi i64 [ 1, %if.then ], [ %mul, %if.end ]
  ret i64 %retval.0
}
