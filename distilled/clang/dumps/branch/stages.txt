== stage_irgen: IR as clang emits it, every local in an alloca ==
define void @main() {
entry:
  %a = alloca i64
  %b = alloca i64
  store i64 3, ptr %a
  store i64 5, ptr %b
  %0 = load i64, ptr %a
  %1 = load i64, ptr %b
  %cmp = icmp slt i64 %0, %1
  br i1 %cmp, label %if.then, label %if.else
if.then:
  %2 = load i64, ptr %b
  call void @print_int(i64 %2)
  br label %if.end
if.else:
  %3 = load i64, ptr %a
  call void @print_int(i64 %3)
  br label %if.end
if.end:
  ret void
}

main:
bb.0.entry:
  -- DAG as built --
  t0: i64 = Constant <3>
  t1: i64 = Constant <5>
  t2: i1 = setcc t0, t1
  t3: ch = brcond t2, %bb.1
  t4: ch = br t3, %bb.2
  -- DAG after selection --
  t0: i64 = MOV64ri32 <3>
  t1: i64 = MOV64ri32 <5>
  t2: i1 = CMP64ri8 t0, t1
  t3: ch = JCC_1 t2, %bb.1
  t4: ch = JMP_1 t3, %bb.2
bb.1.if.then:
  -- DAG as built --
  t0: i64 = Constant <5>   ; result goes to $rdi
  t1: ch = CopyToReg t0, $rdi
  t2: ch = call t1, @print_int
  t3: ch = br t2, %bb.3
  -- DAG after selection --
  t0: i64 = MOV64ri32 <5>   ; result goes to $rdi
  t1: ch = COPY t0, $rdi
  t2: ch = CALL64pcrel32 t1, @print_int
  t3: ch = JMP_1 t2, %bb.3
bb.2.if.else:
  -- DAG as built --
  t0: i64 = Constant <3>   ; result goes to $rdi
